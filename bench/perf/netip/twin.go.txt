package main

import (
	"fmt"
	"net/netip"
	"os"
	"slices"
)

var addrs = []string{"10.1.2.3", "192.168.77.200", "2001:db8::1:2:3", "fe80::a:b:c:d", "::ffff:1.2.3.4", "fd12:3456:789a::1", "8.8.8.8", "2606:4700:4700::1111"}
var v4 = []string{"10.1.2.3", "192.168.77.200", "8.8.8.8", "172.16.254.1"}
var v6 = []string{"2001:db8::1:2:3", "fe80::a:b:c:d", "fd12:3456:789a::1", "2606:4700:4700::1111"}

func main() {
	w := "mixed"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	n := 0
	switch w {
	case "mixed":
		for i := 0; i < 200000; i++ {
			a, _ := netip.ParseAddr(addrs[i%8])
			b, _ := netip.ParseAddr(addrs[(i+3)%8])
			if a.IsPrivate() {
				n++
			}
			n += len(a.String()) + len(b.String())
		}
	case "parse4":
		for i := 0; i < 400000; i++ {
			a, _ := netip.ParseAddr(v4[i%4])
			n += a.BitLen()
		}
	case "parse6":
		for i := 0; i < 400000; i++ {
			a, _ := netip.ParseAddr(v6[i%4])
			n += a.BitLen()
		}
	case "display4", "display6":
		src := v4
		if w == "display6" {
			src = v6
		}
		xs := make([]netip.Addr, 0, 4)
		for _, s := range src {
			a, _ := netip.ParseAddr(s)
			xs = append(xs, a)
		}
		for i := 0; i < 400000; i++ {
			n += len(xs[i%4].String())
		}
	case "compare":
		var xs []netip.Addr
		for i := 0; i < 10000; i++ {
			a, _ := netip.ParseAddr(addrs[i%8])
			xs = append(xs, a)
			b := [16]byte{0x20, 0x01, 0x0d, 0xb8, 14: byte(i / 256), 15: byte(i % 256)}
			xs = append(xs, netip.AddrFrom16(b))
		}
		for r := 0; r < 20; r++ {
			s := slices.Clone(xs)
			slices.SortStableFunc(s, netip.Addr.Compare)
			n += len(s) + s[0].BitLen()
		}
	}
	fmt.Println(w, n)
}
