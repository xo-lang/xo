package main

import (
	"fmt"
	"net/netip"
)

var addrs = []string{"10.1.2.3", "192.168.77.200", "2001:db8::1:2:3", "fe80::a:b:c:d", "::ffff:1.2.3.4", "fd12:3456:789a::1", "8.8.8.8", "2606:4700:4700::1111"}

func main() {
	n := 0
	for i := 0; i < 200000; i++ {
		a, _ := netip.ParseAddr(addrs[i%8])
		b, _ := netip.ParseAddr(addrs[(i+3)%8])
		if a.IsPrivate() {
			n++
		}
		n += len(a.String()) + len(b.String())
	}
	fmt.Println("netip", n)
}
