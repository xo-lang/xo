package main

import (
	"fmt"
	"os"
	"strings"

	"golang.org/x/text/unicode/norm"
)

func text(ascii bool) string {
	var b strings.Builder
	r := 1
	for i := 0; i < 600000; i++ {
		r = (r*1103515245 + 12345) % 2147483648
		k := (r >> 16) % 40
		switch {
		case ascii || k < 36:
			b.WriteRune(rune(32 + (r>>8)%95))
		case k < 37:
			b.WriteRune(rune(0xC0 + (r>>8)%64))
		case k < 38:
			b.WriteRune(rune(97 + (r>>8)%26))
			b.WriteRune(rune(0x300 + (r>>4)%0x30))
		case k < 39:
			b.WriteRune(rune(0xAC00 + (r>>8)%11172))
		default:
			b.WriteRune(rune(0xFB00 + (r>>8)%7))
		}
	}
	return b.String()
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	s := text(w == "ascii")
	n := 0
	if w != "none" {
		for i := 0; i < 20; i++ {
			var t string
			switch w {
			case "nfd":
				t = norm.NFD.String(s)
			case "nfkc":
				t = norm.NFKC.String(s)
			default:
				t = norm.NFC.String(s)
			}
			n = len(t)
		}
	}
	fmt.Printf("%s %d\n", w, n)
}
