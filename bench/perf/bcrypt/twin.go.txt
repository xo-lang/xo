package main

import (
	"fmt"
	"os"

	"golang.org/x/crypto/bcrypt"
)

const H = "$2a$10$XajjQvNhvvRt5GSeFk1xFeyqRrsxkhBkUiQeg0dt.wU1qD4aFDcga"

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	sum := 0
	if w == "hash" {
		for i := 0; i < 20; i++ {
			h, err := bcrypt.GenerateFromPassword([]byte("benchmark password"), 10)
			if err != nil {
				panic(err)
			}
			sum += len(h)
		}
	} else if w == "compare" {
		for i := 0; i < 20; i++ {
			if bcrypt.CompareHashAndPassword([]byte(H), []byte("allmine")) == nil {
				sum++
			}
		}
	}
	fmt.Printf("%s %d\n", w, sum)
}
