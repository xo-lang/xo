package main

import (
	"fmt"
	"net/url"
	"os"
)

var texts = []string{
	"hello world",
	"a/b/c?d=e&f=g",
	"plain-text_with.safe~chars",
	"café 日本語",
	"100% sure: x+y=z",
	"https://example.com/a b?q=1&r=2#frag",
	"",
	"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789",
}

var queries = []string{
	"a=1&b=2&c=3",
	"q=hello+world&page=2&sort=desc",
	"name=Ada%20Lovelace&langs=en&langs=fr&langs=de",
	"x=%E6%97%A5%E6%9C%AC&y=caf%C3%A9",
	"k",
	"a=1&a=2&a=3&b=&=c",
}

var urls = []string{
	"https://user:pass@example.com:8443/a/b%20c/d?x=1&y=two#frag",
	"http://[fe80::1%25en0]:8080/index.html",
	"mailto:someone@example.org",
	"/relative/path/../to/file.txt?q=a+b",
	"https://go.dev/doc/effective_go.html#names",
	"ftp://ftp.example.com/pub/file%2Fname.tar.gz",
}

var refs = []string{
	"../g",
	"g/./h?y=1",
	"//other.example/x",
	"#s",
	"../../../a/b/../c",
	"?only=query",
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	enc := make([]string, 0, len(texts))
	for _, t := range texts {
		enc = append(enc, url.QueryEscape(t))
	}
	parsed := make([]*url.URL, 0, len(urls))
	for _, u := range urls {
		p, err := url.Parse(u)
		if err != nil {
			p = &url.URL{}
		}
		parsed = append(parsed, p)
	}
	base, _ := url.Parse("http://a.example/b/c/d;p?q")
	rels := make([]*url.URL, 0, len(refs))
	for _, r := range refs {
		p, err := url.Parse(r)
		if err != nil {
			p = &url.URL{}
		}
		rels = append(rels, p)
	}
	sum := 0
	for i := 0; i < 200000; i++ {
		switch w {
		case "escape":
			sum += len(url.QueryEscape(texts[i%8])) + len(url.PathEscape(texts[(i+3)%8]))
		case "unescape":
			a, _ := url.QueryUnescape(enc[i%8])
			b, _ := url.PathUnescape(enc[(i+5)%8])
			sum += len(a) + len(b)
		case "query":
			v, err := url.ParseQuery(queries[i%6])
			if err != nil {
				v = url.Values{}
			}
			sum += len(v) + len(v.Get("a")) + len(v["langs"])
		case "encode":
			v := url.Values{}
			v.Add("q", texts[i%8])
			v.Add("page", "2")
			v.Add("tag", texts[(i+1)%8])
			v.Add("tag", "x y")
			sum += len(v.Encode())
		case "parse":
			u, err := url.Parse(urls[i%6])
			if err != nil {
				sum++
			} else {
				sum += len(u.Host) + len(u.Path) + len(u.RawQuery)
			}
		case "string":
			sum += len(parsed[i%6].String())
		case "resolve":
			sum += len(base.ResolveReference(rels[i%6]).Path)
		default:
			sum += len(texts[i%8]) + i%3
		}
	}
	fmt.Printf("%s %d\n", w, sum)
}
