package main

import (
	"fmt"
	"net/url"
	"os"
)

var texts = []string{
	"hello world",
	"a/b/c?d=e&f=g",
	"plain-text_with.safe~chars",
	"café 日本語",
	"100% sure: x+y=z",
	"https://example.com/a b?q=1&r=2#frag",
	"",
	"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789",
}

var queries = []string{
	"a=1&b=2&c=3",
	"q=hello+world&page=2&sort=desc",
	"name=Ada%20Lovelace&langs=en&langs=fr&langs=de",
	"x=%E6%97%A5%E6%9C%AC&y=caf%C3%A9",
	"k",
	"a=1&a=2&a=3&b=&=c",
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	enc := make([]string, 0, len(texts))
	for _, t := range texts {
		enc = append(enc, url.QueryEscape(t))
	}
	sum := 0
	for i := 0; i < 200000; i++ {
		switch w {
		case "escape":
			sum += len(url.QueryEscape(texts[i%8])) + len(url.PathEscape(texts[(i+3)%8]))
		case "unescape":
			a, _ := url.QueryUnescape(enc[i%8])
			b, _ := url.PathUnescape(enc[(i+5)%8])
			sum += len(a) + len(b)
		case "query":
			v, err := url.ParseQuery(queries[i%6])
			if err != nil {
				v = url.Values{}
			}
			sum += len(v) + len(v.Get("a")) + len(v["langs"])
		case "encode":
			v := url.Values{}
			v.Add("q", texts[i%8])
			v.Add("page", "2")
			v.Add("tag", texts[(i+1)%8])
			v.Add("tag", "x y")
			sum += len(v.Encode())
		default:
			sum += len(texts[i%8]) + i%3
		}
	}
	fmt.Printf("%s %d\n", w, sum)
}
