package main

import (
	"fmt"
	"os"
	"regexp"
	"strings"
)

var words = []string{"alpha", "beta", "gamma", "delta", "cache", "entry", "key", "value", "node", "list"}
var pats = []string{`[a-z]+[0-9]*@[a-z]+\.com`, `(\d{4})-(\d{2})-(\d{2})`, `(\w+)=(\w+)`}
var short = []string{"bob42@mail.com", "2026-10-28", "key=value", "no match here", "x@y", "1999-01-02 and more", "a=b c=d", "alpha beta gamma"}

func next(x int) int { return (x*1103515245 + 12345) % 2147483648 }

func clauses() []string {
	var out []string
	r := 7
	for i := 0; i < 1500; i++ {
		var b strings.Builder
		for k := 0; k < 6; k++ {
			r = next(r)
			c := (r >> 8) % 13
			if c == 0 {
				fmt.Fprintf(&b, "%s%d@%s.com", words[(r>>12)%10], (r>>4)%100, words[(r>>16)%10])
			} else if c == 1 {
				fmt.Fprintf(&b, "%d-%d-%d", 1900+(r>>6)%200, 10+(r>>10)%3, 10+(r>>14)%19)
			} else if c == 2 {
				fmt.Fprintf(&b, "%s=%d", words[(r>>12)%10], (r>>4)%1000)
			} else {
				b.WriteString(words[(r>>12)%10])
			}
			b.WriteString(" ")
		}
		b.WriteString(".")
		out = append(out, b.String())
	}
	return out
}

func musts() []*regexp.Regexp {
	var rs []*regexp.Regexp
	for _, p := range pats {
		rs = append(rs, regexp.MustCompile(p))
	}
	return rs
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	cs := clauses()
	text := strings.Join(cs, " ")
	sum := 0
	switch w {
	case "compile":
		for i := 0; i < 20000; i++ {
			for _, p := range pats {
				sum += len(regexp.MustCompile(p).String())
			}
		}
	case "match":
		rs := musts()
		for i := 0; i < 300000; i++ {
			if rs[i%3].MatchString(short[i%8]) {
				sum++
			}
		}
	case "find_all":
		rs := musts()
		for i := 0; i < 20; i++ {
			for _, r := range rs {
				for _, m := range r.FindAllString(text, -1) {
					sum += len(m) + 1
				}
			}
		}
	case "replace_all":
		d := regexp.MustCompile(pats[1])
		kv := regexp.MustCompile(pats[2])
		for i := 0; i < 20; i++ {
			sum += len(d.ReplaceAllString(text, "$3/$2/$1")) + len(kv.ReplaceAllString(text, "$2=$1"))
		}
	case "captures":
		rs := musts()
		for i := 0; i < 3; i++ {
			for _, c := range cs {
				for _, r := range rs {
					if gs := r.FindStringSubmatchIndex(c); gs != nil {
						for g := 0; g < len(gs); g += 2 {
							if gs[g] >= 0 {
								sum += gs[g+1] - gs[g]
							}
							sum++
						}
					}
				}
			}
		}
	default:
		sum = len(text)
	}
	fmt.Printf("%s %d\n", w, sum)
}
