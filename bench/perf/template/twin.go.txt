package main

import (
	"encoding/json"
	"fmt"
	"os"
	"strings"
	"text/template"
)

const src = "<h1>{{.title}}</h1>\n<ul>{{range .items}}\n<li class=\"{{if .done}}done{{else}}todo{{end}}\">{{.name}} x{{.qty}}</li>{{end}}\n</ul>\n{{with .owner}}<p>{{.name}} ({{len $.items}} items)</p>{{end}}\n"

func dataText() string {
	var b strings.Builder
	b.WriteString("{\"title\": \"Orders\", \"owner\": {\"name\": \"Ada\"}, \"items\": [")
	for i := 0; i < 30; i++ {
		if i > 0 {
			b.WriteString(", ")
		}
		fmt.Fprintf(&b, "{\"name\": \"item%d\", \"qty\": %d, \"done\": %v}", i, i%7+1, i%3 == 0)
	}
	b.WriteString("]}")
	return b.String()
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	var data any
	if err := json.Unmarshal([]byte(dataText()), &data); err != nil {
		panic(err)
	}
	t := template.Must(template.New("page").Parse(src))
	sum := 0
	switch w {
	case "parse":
		for i := 0; i < 20000; i++ {
			p := template.Must(template.New("page").Parse(src))
			sum += len(p.Name()) + i%3
		}
	case "render":
		for i := 0; i < 20000; i++ {
			var b strings.Builder
			if err := t.Execute(&b, data); err != nil {
				panic(err)
			}
			out := b.String()
			sum += len(out) + int(out[len(out)-2])
		}
	default:
		for i := 0; i < 20000; i++ {
			sum += len(src) + i%3
		}
	}
	fmt.Printf("%s %d\n", w, sum)
}
