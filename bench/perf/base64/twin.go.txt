package main

import (
	"encoding/base64"
	"fmt"
	"os"
)

func next(x int) int { return (x*1103515245 + 12345) % 2147483648 }

func inputs() []string {
	var out []string
	r := 12345
	for k := 0; k < 2000; k++ {
		r = next(r)
		n := (r >> 8) % 3001
		b := make([]byte, 0, n)
		for i := 0; i < n; i++ {
			r = next(r)
			b = append(b, byte('0'+(r>>8)%10))
		}
		out = append(out, string(b))
	}
	return out
}

func main() {
	w := "encode"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	ins := inputs()
	sum := 0
	if w == "encode" {
		for p := 0; p < 20; p++ {
			for _, s := range ins {
				e := base64.StdEncoding.EncodeToString([]byte(s))
				sum += len(e)
				if len(e) > 0 {
					sum += int(e[len(e)-1])
				}
			}
		}
	} else if w == "decode" {
		var encoded []string
		for _, s := range ins {
			encoded = append(encoded, base64.StdEncoding.EncodeToString([]byte(s)))
		}
		for p := 0; p < 20; p++ {
			for _, e := range encoded {
				d, err := base64.StdEncoding.DecodeString(e)
				if err != nil {
					panic(err)
				}
				sum += len(d)
				if len(d) > 0 {
					sum += int(d[len(d)-1])
				}
			}
		}
	}
	fmt.Println(w, sum)
}
