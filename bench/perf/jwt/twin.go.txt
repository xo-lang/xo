type Claims struct {
	Sub string `json:"sub"`
	Exp int64  `json:"exp"`
	Iss string `json:"iss"`
}

func (c Claims) GetExpirationTime() (*jwt.NumericDate, error) {
	return jwt.NewNumericDate(time.Unix(c.Exp, 0)), nil
}
func (Claims) GetIssuedAt() (*jwt.NumericDate, error)  { return nil, nil }
func (Claims) GetNotBefore() (*jwt.NumericDate, error) { return nil, nil }
func (c Claims) GetIssuer() (string, error)            { return c.Iss, nil }
func (c Claims) GetSubject() (string, error)           { return c.Sub, nil }
func (Claims) GetAudience() (jwt.ClaimStrings, error)  { return nil, nil }

func priv(s string) crypto.PrivateKey {
	b, _ := pem.Decode([]byte(s))
	k, err := x509.ParsePKCS8PrivateKey(b.Bytes)
	if err != nil {
		panic(err)
	}
	return k
}

func pub(s string) crypto.PublicKey {
	b, _ := pem.Decode([]byte(s))
	k, err := x509.ParsePKIXPublicKey(b.Bytes)
	if err != nil {
		panic(err)
	}
	return k
}

var _ = ecdsa.PublicKey{}
var _ = ed25519.PublicKey{}
var _ = rsa.PublicKey{}

func sign(m jwt.SigningMethod, k any, c Claims) string {
	s, err := jwt.NewWithClaims(m, c).SignedString(k)
	if err != nil {
		panic(err)
	}
	return s
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	c := Claims{Sub: "ann", Exp: 4000000000, Iss: "me"}
	hs := []byte("secret-for-jwt-bench-0123456789abcdef")
	rsaK, p256K, edK := priv(rsaPriv), priv(p256Priv), priv(ed25519Priv)
	rsaP, p256P, edP := pub(rsaPub), pub(p256Pub), pub(ed25519Pub)
	hsTok := sign(jwt.SigningMethodHS256, hs, c)
	rsTok := sign(jwt.SigningMethodRS256, rsaK, c)
	esTok := sign(jwt.SigningMethodES256, p256K, c)
	edTok := sign(jwt.SigningMethodEdDSA, edK, c)
	ok := 0
	now := func() time.Time { return time.Unix(1000, 0) }
	if w == "hs256_sign" {
		for i := 0; i < 20000; i++ {
			if _, err := jwt.NewWithClaims(jwt.SigningMethodHS256, c).SignedString(hs); err == nil {
				ok++
			}
		}
	} else if w != "none" {
		alg, tok, key := "EdDSA", edTok, any(edP)
		switch w {
		case "hs256_verify":
			alg, tok, key = "HS256", hsTok, hs
		case "rs256_verify":
			alg, tok, key = "RS256", rsTok, rsaP
		case "es256_verify":
			alg, tok, key = "ES256", esTok, p256P
		}
		p := jwt.NewParser(jwt.WithValidMethods([]string{alg}), jwt.WithTimeFunc(now), jwt.WithExpirationRequired())
		for i := 0; i < 2000; i++ {
			var got Claims
			if _, err := p.ParseWithClaims(tok, &got, func(*jwt.Token) (any, error) { return key, nil }); err == nil {
				ok++
			}
		}
	}
	fmt.Println(w, ok)
}
