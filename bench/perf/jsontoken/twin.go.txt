package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"os"
)

type Rec struct {
	Key    string   `json:"key"`
	Size   int      `json:"size"`
	Ratio  float64  `json:"ratio"`
	Etag   string   `json:"etag"`
	Public bool     `json:"public"`
	Tags   []string `json:"tags"`
}

func weight(t json.Token) int {
	switch v := t.(type) {
	case json.Delim:
		switch v {
		case '{':
			return 1
		case '}':
			return 2
		case '[':
			return 3
		}
		return 4
	case string:
		return len(v)
	case json.Number:
		return len(v)
	case bool:
		if v {
			return 5
		}
		return 6
	}
	return 7
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	var rs []Rec
	for i := 0; i < 40; i++ {
		rs = append(rs, Rec{Key: fmt.Sprintf("photos/2026/img-%d.jpg", i), Size: i * 7919 % 100003, Ratio: float64(i) * 0.25, Etag: fmt.Sprintf("d41d8cd98f00b204e9800998ecf8427e%d", i), Public: i%3 == 0, Tags: []string{fmt.Sprintf("a%d", i%5), fmt.Sprintf("b<%d>", i%7), "plain"}})
	}
	doc, _ := json.Marshal(rs)
	sum := 0
	for i := 0; i < 8000; i++ {
		switch w {
		case "read":
			dec := json.NewDecoder(bytes.NewReader(doc))
			dec.UseNumber()
			for {
				t, err := dec.Token()
				if err == io.EOF {
					break
				}
				if err != nil {
					panic(err)
				}
				sum += weight(t)
			}
			sum += i % 3
		case "write":
			out, _ := json.Marshal(rs)
			sum += len(out) + int(out[i%len(out)])
		default:
			sum += len(doc) + i%3
		}
	}
	fmt.Println(w, sum)
}
