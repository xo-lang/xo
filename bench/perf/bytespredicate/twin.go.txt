// The Go twin of bench/perf/bytespredicate/main.xo: the same texts and loop
// counts through bytes.IndexFunc, LastIndexFunc, ContainsFunc, TrimFunc,
// TrimLeftFunc, TrimRightFunc, FieldsFunc and Map, printing the same
// "<workload> <checksum>" line. It is a .txt so the Go tool does not build it
// as part of the repository; copy it out and `go build` it.
package main

import (
	"bytes"
	"fmt"
	"os"
	"unicode"
	"unicode/utf8"
)

const n = 200000

var texts = [][]byte{
	[]byte("  the quick brown fox, jumps over the lazy dog  "),
	[]byte("1,2,3,4"),
	[]byte("  héllo, wörld, ça va, 日本語 テキスト   "),
	[]byte("  \t  mixed ascii and ✓ unicode ✓  "),
	[]byte("abcabcabcabcabcabcabc"),
	[]byte("a\xffb, c\x80 ,"),
}

func isFFFD(r rune) bool { return r == utf8.RuneError }

func rot13(r rune) rune {
	switch {
	case r >= 'a' && r <= 'z':
		return 'a' + (r-'a'+13)%26
	case r >= 'A' && r <= 'Z':
		return 'A' + (r-'A'+13)%26
	}
	return r
}

func indexAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += len(texts[i%len(texts)])
	}
	return sum
}

func findWhereAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += bytes.IndexFunc(texts[i%len(texts)], unicode.IsSpace)
	}
	return sum
}

func rfindWhereAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += bytes.LastIndexFunc(texts[i%len(texts)], unicode.IsSpace)
	}
	return sum
}

func containsWhereAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		if bytes.ContainsFunc(texts[i%len(texts)], isFFFD) {
			sum++
		}
	}
	return sum
}

func trimWhereAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += len(bytes.TrimFunc(texts[i%len(texts)], unicode.IsSpace))
	}
	return sum
}

func trimStartWhereAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += len(bytes.TrimLeftFunc(texts[i%len(texts)], unicode.IsLetter))
	}
	return sum
}

func trimEndWhereAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += len(bytes.TrimRightFunc(texts[i%len(texts)], isFFFD))
	}
	return sum
}

func splitWhereAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range bytes.FieldsFunc(texts[i%len(texts)], unicode.IsSpace) {
			sum += len(p)
		}
	}
	return sum
}

func mapRunesAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += len(bytes.Map(rot13, texts[i%len(texts)]))
	}
	return sum
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	sum := 0
	switch w {
	case "find_where":
		sum = findWhereAll()
	case "rfind_where":
		sum = rfindWhereAll()
	case "contains_where":
		sum = containsWhereAll()
	case "trim_where":
		sum = trimWhereAll()
	case "trim_start_where":
		sum = trimStartWhereAll()
	case "trim_end_where":
		sum = trimEndWhereAll()
	case "split_where":
		sum = splitWhereAll()
	case "map_runes":
		sum = mapRunesAll()
	case "index":
		sum = indexAll()
	}
	fmt.Printf("%s %d\n", w, sum)
}
