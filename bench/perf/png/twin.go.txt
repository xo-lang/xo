package main

import (
	"bytes"
	"fmt"
	"image"
	"image/color"
	"image/png"
	"os"
)

func next(x int) int { return (x*1103515245 + 12345) % 2147483648 }

func make4() []image.Image {
	var out []image.Image
	rgba := image.NewRGBA(image.Rect(0, 0, 256, 256))
	nrgba := image.NewNRGBA(image.Rect(0, 0, 256, 256))
	gray := image.NewGray(image.Rect(0, 0, 256, 256))
	var pal color.Palette
	for i := 0; i < 16; i++ {
		pal = append(pal, color.NRGBA{uint8(i * 16), uint8(255 - i*16), uint8(i * 7), 255})
	}
	pm := image.NewPaletted(image.Rect(0, 0, 256, 256), pal)
	r := 1
	for y := 0; y < 256; y++ {
		for x := 0; x < 256; x++ {
			r = next(r)
			noise := (r >> 8) % 5
			i := rgba.PixOffset(x, y)
			rgba.Pix[i], rgba.Pix[i+1], rgba.Pix[i+2], rgba.Pix[i+3] = uint8((x+noise)%256), uint8((y+noise)%256), uint8((x+y)/2%256), 255
		}
	}
	r = 2
	for y := 0; y < 256; y++ {
		for x := 0; x < 256; x++ {
			r = next(r)
			noise := (r >> 8) % 5
			a := 128 + (x+y)/4%128
			i := nrgba.PixOffset(x, y)
			nrgba.Pix[i], nrgba.Pix[i+1], nrgba.Pix[i+2], nrgba.Pix[i+3] = uint8((x+noise)%a), uint8((y+noise)%a), uint8((x+y)/2%a), uint8(a)
		}
	}
	r = 3
	for y := 0; y < 256; y++ {
		for x := 0; x < 256; x++ {
			r = next(r)
			noise := (r >> 8) % 5
			gray.Pix[gray.PixOffset(x, y)] = uint8((x + y + noise) % 256)
		}
	}
	r = 4
	for y := 0; y < 256; y++ {
		for x := 0; x < 256; x++ {
			r = next(r)
			noise := (r >> 8) % 5
			e := 0
			if noise == 0 {
				e = 1
			}
			pm.Pix[pm.PixOffset(x, y)] = uint8((x/16 + y/16 + e) % 16)
		}
	}
	out = append(out, rgba, nrgba, gray, pm)
	return out
}

func main() {
	w := "encode"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	imgs := make4()
	var files [][]byte
	sum := 0
	for _, im := range imgs {
		var b bytes.Buffer
		png.Encode(&b, im)
		files = append(files, b.Bytes())
		sum += b.Len()
	}
	if w == "encode" {
		for p := 0; p < 8; p++ {
			for _, im := range imgs {
				var b bytes.Buffer
				png.Encode(&b, im)
				e := b.Bytes()
				sum += len(e) + int(e[len(e)-1])
			}
		}
	} else if w == "decode" {
		for p := 0; p < 8; p++ {
			for _, f := range files {
				im, err := png.Decode(bytes.NewReader(f))
				if err != nil {
					continue
				}
				b := im.Bounds()
				var last byte
				switch m := im.(type) {
				case *image.RGBA:
					last = m.Pix[len(m.Pix)-1]
				case *image.NRGBA:
					last = m.Pix[len(m.Pix)-1]
				case *image.Gray:
					last = m.Pix[len(m.Pix)-1]
				case *image.Paletted:
					last = m.Pix[len(m.Pix)-1]
				}
				sum += b.Dx()*b.Dy() + int(last)
			}
		}
	}
	fmt.Println(w, sum)
}
