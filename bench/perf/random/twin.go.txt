// The Go twin of main.xo: math/rand/v2's NormFloat64, ExpFloat64, Perm
// and Zipf from the top-level generator.
package main

import (
	"fmt"
	"math/rand/v2"
	"os"
)

func share(k, n int) string {
	c := (k*100 + n/2) / n
	return fmt.Sprintf("0.%02d", c)
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	n := 1000000
	switch w {
	case "normal":
		k := 0
		for i := 0; i < n; i++ {
			x := rand.NormFloat64()
			if x > -1 && x < 1 {
				k++
			}
		}
		fmt.Printf("normal %d %s\n", n, share(k, n))
	case "exp":
		k := 0
		for i := 0; i < n; i++ {
			if rand.ExpFloat64() < 1 {
				k++
			}
		}
		fmt.Printf("exp %d %s\n", n, share(k, n))
	case "perm":
		k := 0
		for i := 0; i < 2000; i++ {
			if rand.Perm(1000)[0] < 500 {
				k++
			}
		}
		fmt.Printf("perm %d %s\n", 2000*1000, share(k*10, 20000))
	case "zipf":
		z := rand.NewZipf(rand.New(rand.NewChaCha8([32]byte{1})), 1.5, 1, 1000)
		k := 0
		for i := 0; i < n; i++ {
			if z.Uint64() == 0 {
				k++
			}
		}
		fmt.Printf("zipf %d %s\n", n, share(k, n))
	default:
		fmt.Println("none 0 0.00")
	}
}
