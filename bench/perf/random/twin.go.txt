// The Go twin of main.xo: math/rand/v2's NormFloat64, ExpFloat64, Perm
// and Zipf from the top-level generator, and its bare Uint64 words
// (draw).
package main

import (
	"fmt"
	"math"
	"math/rand/v2"
	"os"
)

func share(k, n int) string {
	c := (k*100 + n/2) / n
	return fmt.Sprintf("0.%02d", c)
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	n := 1000000
	switch w {
	case "normal":
		k := 0
		for i := 0; i < n; i++ {
			x := rand.NormFloat64()
			if x > -1 && x < 1 {
				k++
			}
		}
		fmt.Printf("normal %d %s\n", n, share(k, n))
	case "exp":
		k := 0
		for i := 0; i < n; i++ {
			if rand.ExpFloat64() < 1 {
				k++
			}
		}
		fmt.Printf("exp %d %s\n", n, share(k, n))
	case "perm":
		k := 0
		for i := 0; i < 2000; i++ {
			if rand.Perm(1000)[0] < 500 {
				k++
			}
		}
		fmt.Printf("perm %d %s\n", 2000*1000, share(k*10, 20000))
	case "zipf":
		if len(os.Args) <= 2 || os.Args[2] != "pkg" {
			// Go's Zipf (math/rand/v2 zipf.go) on Go's portable Exp and
			// Log (decision 0721 sizing): package math runs exp_amd64.s
			// and log_amd64.s on amd64; "zipf pkg" runs rand.Zipf.
			z := newPZipf(rand.New(rand.NewChaCha8([32]byte{1})), 1.5, 1, 1000)
			k := 0
			for i := 0; i < n; i++ {
				if z.uint64() == 0 {
					k++
				}
			}
			fmt.Printf("zipf %d %s\n", n, share(k, n))
			return
		}
		z := rand.NewZipf(rand.New(rand.NewChaCha8([32]byte{1})), 1.5, 1, 1000)
		k := 0
		for i := 0; i < n; i++ {
			if z.Uint64() == 0 {
				k++
			}
		}
		fmt.Printf("zipf %d %s\n", n, share(k, n))
	case "draw":
		// the words' low bits counted, so every word is used
		k := 0
		for i := 0; i < 50*n; i++ {
			k += int(rand.Uint64() & 1)
		}
		fmt.Printf("draw %d %s\n", 50*n, share(k, 50*n))
	default:
		fmt.Println("none 0 0.00")
	}
}

// pZipf is math/rand/v2's Zipf (go1.26 zipf.go) with gmExp and gmLog.
type pZipf struct {
	r                       *rand.Rand
	imax, v, q, s           float64
	oneminusQ, oneminusQinv float64
	hxm, hx0minusHxm        float64
}

func (z *pZipf) h(x float64) float64 {
	return gmExp(z.oneminusQ*gmLog(z.v+x)) * z.oneminusQinv
}

func (z *pZipf) hinv(x float64) float64 {
	return gmExp(z.oneminusQinv*gmLog(z.oneminusQ*x)) - z.v
}

func newPZipf(r *rand.Rand, s float64, v float64, imax uint64) *pZipf {
	z := new(pZipf)
	z.r = r
	z.imax = float64(imax)
	z.v = v
	z.q = s
	z.oneminusQ = 1.0 - z.q
	z.oneminusQinv = 1.0 / z.oneminusQ
	z.hxm = z.h(z.imax + 0.5)
	z.hx0minusHxm = z.h(0.5) - gmExp(gmLog(z.v)*(-z.q)) - z.hxm
	z.s = 1 - z.hinv(z.h(1.5)-gmExp(-z.q*gmLog(z.v+1.0)))
	return z
}

func (z *pZipf) uint64() uint64 {
	k := 0.0
	for {
		r := z.r.Float64()
		ur := z.hxm + r*z.hx0minusHxm
		x := z.hinv(ur)
		k = math.Floor(x + 0.5)
		if k-x <= z.s {
			break
		}
		if ur >= z.h(k+0.5)-gmExp(-gmLog(k+z.v)*z.q) {
			break
		}
	}
	return uint64(k)
}

// Go's portable math for erf, erfc, exp and pow and what they call, from
// std/rt/gomath_gen.go (scripts/gomath; decision 0287) with package math's
// Sqrt and Trunc.

func gmAbs(x float64) float64 {
	return gmFloat64frombits(gmFloat64bits(x) &^ (1 << 63))
}

const (
	gmuvnan    = 0x7FF8000000000001
	gmuvinf    = 0x7FF0000000000000
	gmuvneginf = 0xFFF0000000000000
	gmuvone    = 0x3FF0000000000000
	gmmask     = 0x7FF
	gmshift    = 64 - 11 - 1
	gmbias     = 1023
	gmsignMask = 1 << 63
	gmfracMask = 1<<gmshift - 1
)

func gmInf(sign int) float64 {
	var v uint64
	if sign >= 0 {
		v = gmuvinf
	} else {
		v = gmuvneginf
	}
	return gmFloat64frombits(v)
}

func gmNaN() float64 { return gmFloat64frombits(gmuvnan) }

func gmIsNaN(f float64) (is bool) {

	return f != f
}

func gmIsInf(f float64, sign int) bool {

	if sign == 0 {
		f = gmAbs(f)
	} else if sign < 0 {
		f = -f
	}
	return f > gmMaxFloat64
}

func gmnormalize(x float64) (y float64, exp int) {
	const SmallestNormal = 2.2250738585072014e-308
	if gmAbs(x) < SmallestNormal {
		return float64(x * (1 << 52)), -52
	}
	return x, 0
}

func gmCopysign(f, sign float64) float64 {
	const signBit = 1 << 63
	return gmFloat64frombits(gmFloat64bits(f)&^signBit | gmFloat64bits(sign)&signBit)
}

func gmerf(x float64) float64 {
	const (
		VeryTiny = 2.848094538889218e-306
		Small    = 1.0 / (1 << 28)
	)

	switch {
	case gmIsNaN(x):
		return gmNaN()
	case gmIsInf(x, 1):
		return 1
	case gmIsInf(x, -1):
		return -1
	}
	sign := false
	if x < 0 {
		x = -x
		sign = true
	}
	if x < 0.84375 {
		var temp float64
		if x < Small {
			if x < VeryTiny {
				temp = float64(0.125 * (float64(8.0*x) + float64(gmefx8*x)))
			} else {
				temp = x + float64(gmefx*x)
			}
		} else {
			z := float64(x * x)
			r := gmpp0 + float64(z*(gmpp1+float64(z*(gmpp2+float64(z*(gmpp3+float64(z*gmpp4)))))))
			s := 1 + float64(z*(gmqq1+float64(z*(gmqq2+float64(z*(gmqq3+float64(z*(gmqq4+float64(z*gmqq5)))))))))
			y := r / s
			temp = x + float64(x*y)
		}
		if sign {
			return -temp
		}
		return temp
	}
	if x < 1.25 {
		s := x - 1
		P := gmpa0 + float64(s*(gmpa1+float64(s*(gmpa2+float64(s*(gmpa3+float64(s*(gmpa4+float64(s*(gmpa5+float64(s*gmpa6)))))))))))
		Q := 1 + float64(s*(gmqa1+float64(s*(gmqa2+float64(s*(gmqa3+float64(s*(gmqa4+float64(s*(gmqa5+float64(s*gmqa6)))))))))))
		if sign {
			return -gmerx - P/Q
		}
		return gmerx + P/Q
	}
	if x >= 6 {
		if sign {
			return -1
		}
		return 1
	}
	s := 1 / (float64(x * x))
	var R, S float64
	if x < 1/0.35 {
		R = gmra0 + float64(s*(gmra1+float64(s*(gmra2+float64(s*(gmra3+float64(s*(gmra4+float64(s*(gmra5+float64(s*(gmra6+float64(s*gmra7)))))))))))))
		S = 1 + float64(s*(gmsa1+float64(s*(gmsa2+float64(s*(gmsa3+float64(s*(gmsa4+float64(s*(gmsa5+float64(s*(gmsa6+float64(s*(gmsa7+float64(s*gmsa8)))))))))))))))
	} else {
		R = gmrb0 + float64(s*(gmrb1+float64(s*(gmrb2+float64(s*(gmrb3+float64(s*(gmrb4+float64(s*(gmrb5+float64(s*gmrb6)))))))))))
		S = 1 + float64(s*(gmsb1+float64(s*(gmsb2+float64(s*(gmsb3+float64(s*(gmsb4+float64(s*(gmsb5+float64(s*(gmsb6+float64(s*gmsb7)))))))))))))
	}
	z := gmFloat64frombits(gmFloat64bits(x) & 0xffffffff00000000)
	r := float64(gmExp(float64(-z*z)-0.5625) * gmExp(float64((z-x)*(z+x))+R/S))
	if sign {
		return r/x - 1
	}
	return 1 - r/x
}

func gmerfc(x float64) float64 {
	const Tiny = 1.0 / (1 << 56)

	switch {
	case gmIsNaN(x):
		return gmNaN()
	case gmIsInf(x, 1):
		return 0
	case gmIsInf(x, -1):
		return 2
	}
	sign := false
	if x < 0 {
		x = -x
		sign = true
	}
	if x < 0.84375 {
		var temp float64
		if x < Tiny {
			temp = x
		} else {
			z := float64(x * x)
			r := gmpp0 + float64(z*(gmpp1+float64(z*(gmpp2+float64(z*(gmpp3+float64(z*gmpp4)))))))
			s := 1 + float64(z*(gmqq1+float64(z*(gmqq2+float64(z*(gmqq3+float64(z*(gmqq4+float64(z*gmqq5)))))))))
			y := r / s
			if x < 0.25 {
				temp = x + float64(x*y)
			} else {
				temp = 0.5 + (float64(x*y) + (x - 0.5))
			}
		}
		if sign {
			return 1 + temp
		}
		return 1 - temp
	}
	if x < 1.25 {
		s := x - 1
		P := gmpa0 + float64(s*(gmpa1+float64(s*(gmpa2+float64(s*(gmpa3+float64(s*(gmpa4+float64(s*(gmpa5+float64(s*gmpa6)))))))))))
		Q := 1 + float64(s*(gmqa1+float64(s*(gmqa2+float64(s*(gmqa3+float64(s*(gmqa4+float64(s*(gmqa5+float64(s*gmqa6)))))))))))
		if sign {
			return 1 + gmerx + P/Q
		}
		return 1 - gmerx - P/Q

	}
	if x < 28 {
		s := 1 / (float64(x * x))
		var R, S float64
		if x < 1/0.35 {
			R = gmra0 + float64(s*(gmra1+float64(s*(gmra2+float64(s*(gmra3+float64(s*(gmra4+float64(s*(gmra5+float64(s*(gmra6+float64(s*gmra7)))))))))))))
			S = 1 + float64(s*(gmsa1+float64(s*(gmsa2+float64(s*(gmsa3+float64(s*(gmsa4+float64(s*(gmsa5+float64(s*(gmsa6+float64(s*(gmsa7+float64(s*gmsa8)))))))))))))))
		} else {
			if sign && x > 6 {
				return 2
			}
			R = gmrb0 + float64(s*(gmrb1+float64(s*(gmrb2+float64(s*(gmrb3+float64(s*(gmrb4+float64(s*(gmrb5+float64(s*gmrb6)))))))))))
			S = 1 + float64(s*(gmsb1+float64(s*(gmsb2+float64(s*(gmsb3+float64(s*(gmsb4+float64(s*(gmsb5+float64(s*(gmsb6+float64(s*gmsb7)))))))))))))
		}
		z := gmFloat64frombits(gmFloat64bits(x) & 0xffffffff00000000)
		r := float64(gmExp(float64(-z*z)-0.5625) * gmExp(float64((z-x)*(z+x))+R/S))
		if sign {
			return 2 - r/x
		}
		return r / x
	}
	if sign {
		return 2
	}
	return 0
}

func gmexp(x float64) float64 {
	const (
		Ln2Hi = 6.93147180369123816490e-01
		Ln2Lo = 1.90821492927058770002e-10
		Log2e = 1.44269504088896338700e+00

		Overflow  = 7.09782712893383973096e+02
		Underflow = -7.45133219101941108420e+02
		NearZero  = 1.0 / (1 << 28)
	)

	switch {
	case gmIsNaN(x):
		return x
	case x > Overflow:
		return gmInf(1)
	case x < Underflow:
		return 0
	case -NearZero < x && x < NearZero:
		return 1 + x
	}

	var k int
	switch {
	case x < 0:
		k = int(float64(Log2e*x) - 0.5)
	case x > 0:
		k = int(float64(Log2e*x) + 0.5)
	}
	hi := x - float64(float64(k)*Ln2Hi)
	lo := float64(float64(k) * Ln2Lo)

	return gmexpmulti(hi, lo, k)
}

func gmexpmulti(hi, lo float64, k int) float64 {
	const (
		P1 = 1.66666666666666657415e-01
		P2 = -2.77777777770155933842e-03
		P3 = 6.61375632143793436117e-05
		P4 = -1.65339022054652515390e-06
		P5 = 4.13813679705723846039e-08
	)

	r := hi - lo
	t := float64(r * r)
	c := r - float64(t*(P1+float64(t*(P2+float64(t*(P3+float64(t*(P4+float64(t*P5)))))))))
	y := 1 - ((lo - (float64(r*c))/(2-c)) - hi)

	return gmLdexp(y, k)
}

func gmfrexp(f float64) (frac float64, exp int) {

	switch {
	case f == 0:
		return f, 0
	case gmIsInf(f, 0) || gmIsNaN(f):
		return f, 0
	}
	f, exp = gmnormalize(f)
	x := gmFloat64bits(f)
	exp += int((x>>gmshift)&gmmask) - gmbias + 1
	x &^= gmmask << gmshift
	x |= (-1 + gmbias) << gmshift
	frac = gmFloat64frombits(x)
	return
}

func gmLdexp(frac float64, exp int) float64 {
	if gmhaveArchLdexp {
		return gmarchLdexp(frac, exp)
	}
	return gmldexp(frac, exp)
}

func gmldexp(frac float64, exp int) float64 {

	switch {
	case frac == 0:
		return frac
	case gmIsInf(frac, 0) || gmIsNaN(frac):
		return frac
	}
	frac, e := gmnormalize(frac)
	exp += e
	x := gmFloat64bits(frac)
	exp += int(x>>gmshift)&gmmask - gmbias
	if exp < -1075 {
		return gmCopysign(0, frac)
	}
	if exp > 1023 {
		if frac < 0 {
			return gmInf(-1)
		}
		return gmInf(1)
	}
	var m float64 = 1
	if exp < -1022 {
		exp += 53
		m = 1.0 / (1 << 53)
	}
	x &^= gmmask << gmshift
	x |= uint64(exp+gmbias) << gmshift
	return float64(m * gmFloat64frombits(x))
}

func gmlog(x float64) float64 {
	const (
		Ln2Hi = 6.93147180369123816490e-01
		Ln2Lo = 1.90821492927058770002e-10
		L1    = 6.666666666666735130e-01
		L2    = 3.999999999940941908e-01
		L3    = 2.857142874366239149e-01
		L4    = 2.222219843214978396e-01
		L5    = 1.818357216161805012e-01
		L6    = 1.531383769920937332e-01
		L7    = 1.479819860511658591e-01
	)

	switch {
	case gmIsNaN(x) || gmIsInf(x, 1):
		return x
	case x < 0:
		return gmNaN()
	case x == 0:
		return gmInf(-1)
	}

	f1, ki := gmFrexp(x)
	if f1 < gmSqrt2/2 {
		f1 *= 2
		ki--
	}
	f := f1 - 1
	k := float64(ki)

	s := f / (2 + f)
	s2 := float64(s * s)
	s4 := float64(s2 * s2)
	t1 := float64(s2 * (L1 + float64(s4*(L3+float64(s4*(L5+float64(s4*L7)))))))
	t2 := float64(s4 * (L2 + float64(s4*(L4+float64(s4*L6)))))
	R := t1 + t2
	hfsq := float64(float64(0.5*f) * f)
	return float64(k*Ln2Hi) - ((hfsq - (float64(s*(hfsq+R)) + float64(k*Ln2Lo))) - f)
}

func gmModf(f float64) (integer float64, fractional float64) {
	integer = math.Trunc(f)
	fractional = gmCopysign(f-integer, f)
	return
}

func gmisOddInt(x float64) bool {
	if gmAbs(x) >= (1 << 53) {

		return false
	}

	xi, xf := gmModf(x)
	return xf == 0 && int64(xi)&1 == 1
}

func gmPow(x, y float64) float64 {
	if gmhaveArchPow {
		return gmarchPow(x, y)
	}
	return gmpow(x, y)
}

func gmpow(x, y float64) float64 {
	switch {
	case y == 0 || x == 1:
		return 1
	case y == 1:
		return x
	case gmIsNaN(x) || gmIsNaN(y):
		return gmNaN()
	case x == 0:
		switch {
		case y < 0:
			if gmSignbit(x) && gmisOddInt(y) {
				return gmInf(-1)
			}
			return gmInf(1)
		case y > 0:
			if gmSignbit(x) && gmisOddInt(y) {
				return x
			}
			return 0
		}
	case gmIsInf(y, 0):
		switch {
		case x == -1:
			return 1
		case (gmAbs(x) < 1) == gmIsInf(y, 1):
			return 0
		default:
			return gmInf(1)
		}
	case gmIsInf(x, 0):
		if gmIsInf(x, -1) {
			return gmPow(1/x, -y)
		}
		switch {
		case y < 0:
			return 0
		case y > 0:
			return gmInf(1)
		}
	case y == 0.5:
		return math.Sqrt(x)
	case y == -0.5:
		return 1 / math.Sqrt(x)
	}

	yi, yf := gmModf(gmAbs(y))
	if yf != 0 && x < 0 {
		return gmNaN()
	}
	if yi >= 1<<63 {

		switch {
		case x == -1:
			return 1
		case (gmAbs(x) < 1) == (y > 0):
			return 0
		default:
			return gmInf(1)
		}
	}

	a1 := 1.0
	ae := 0

	if yf != 0 {
		if yf > 0.5 {
			yf--
			yi++
		}
		a1 = gmExp(float64(yf * gmLog(x)))
	}

	x1, xe := gmFrexp(x)
	for i := int64(yi); i != 0; i >>= 1 {
		if xe < -1<<12 || 1<<12 < xe {

			ae += xe
			break
		}
		if i&1 == 1 {
			a1 *= x1
			ae += xe
		}
		x1 *= x1
		xe <<= 1
		if x1 < .5 {
			x1 += x1
			xe--
		}
	}

	if y < 0 {
		a1 = 1 / a1
		ae = -ae
	}
	return gmLdexp(a1, ae)
}

func gmSignbit(x float64) bool {
	return int64(gmFloat64bits(x)) < 0
}
