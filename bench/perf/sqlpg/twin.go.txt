package main

import (
	"context"
	"fmt"
	"os"

	"github.com/jackc/pgx/v5"
)

func main() {
	w := "scan"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	ctx := context.Background()
	conn, err := pgx.Connect(ctx, os.Getenv("PG_URL"))
	if err != nil {
		panic(err)
	}
	defer conn.Close(ctx)
	sum := 0
	if w == "setup" {
		for _, q := range []string{
			"DROP TABLE IF EXISTS items",
			"CREATE TABLE items (id INT, name TEXT, score DOUBLE PRECISION, active BOOL, n INT)",
			"INSERT INTO items SELECT i, 'item-' || i, i * 0.5, i % 3 = 0, i * 7 % 101 FROM generate_series(0, 999) AS i",
		} {
			if _, err := conn.Exec(ctx, q); err != nil {
				panic(err)
			}
		}
	} else {
		for k := 0; k < 300; k++ {
			rs, err := conn.Query(ctx, "SELECT id, name, score, active, n FROM items ORDER BY id")
			if err != nil {
				panic(err)
			}
			for rs.Next() {
				var id, n int32
				var name string
				var score float64
				var active bool
				if err := rs.Scan(&id, &name, &score, &active, &n); err != nil {
					panic(err)
				}
				a := 0
				if active {
					a = 1
				}
				sum += int(id) + len(name) + int(score) + a + int(n)
			}
			rs.Close()
			if rs.Err() != nil {
				panic(rs.Err())
			}
		}
	}
	fmt.Println(w, sum)
}
