package main

import (
	"fmt"
	"hash/crc32"
	"os"
)

func buffer(n int) []byte {
	b := make([]byte, n)
	x := 12345
	for i := range b {
		x = (x*1103515245 + 12345) % 2147483648
		b[i] = byte((x >> 16) & 255)
	}
	return b
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	big := buffer(65536)
	keys := buffer(4096)
	sum := 0
	castagnoli := crc32.MakeTable(crc32.Castagnoli)
	switch w {
	case "crc32", "crc32c":
		t := crc32.IEEETable
		if w == "crc32c" {
			t = castagnoli
		}
		for i := 0; i < 2000; i++ {
			sum = sum ^ int(crc32.Checksum(big, t)) ^ i
		}
	case "keys", "keysc":
		t := crc32.IEEETable
		if w == "keysc" {
			t = castagnoli
		}
		for i := 0; i < 2000000; i++ {
			at := i % 4000
			sum = sum ^ int(crc32.Checksum(keys[at:at+8+i%33], t))
		}
	case "update":
		for i := 0; i < 2000; i++ {
			var c uint32
			for at := 0; at < len(big); at += 1024 {
				c = crc32.Update(c, crc32.IEEETable, big[at:at+1024])
			}
			sum = sum ^ int(c) ^ i
		}
	default:
		sum = len(big) + len(keys)
	}
	fmt.Printf("%s %d\n", w, sum)
}
