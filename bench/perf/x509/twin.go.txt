func parsePEM(b []byte) []*x509.Certificate {
	var out []*x509.Certificate
	for {
		p, rest := pem.Decode(b)
		if p == nil {
			return out
		}
		b = rest
		if p.Type != "CERTIFICATE" {
			continue
		}
		c, err := x509.ParseCertificate(p.Bytes)
		if err != nil {
			panic(err)
		}
		out = append(out, c)
	}
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	chain := []byte(leaf + "\n" + inter + "\n" + root + "\n")
	s := 0
	switch w {
	case "parse":
		for i := 0; i < 20000; i++ {
			cs := parsePEM(chain)
			s += len(cs) + len(cs[0].DNSNames)
		}
	case "verify":
		cs := parsePEM(chain)
		roots := x509.NewCertPool()
		roots.AddCert(cs[2])
		inters := x509.NewCertPool()
		inters.AddCert(cs[1])
		for i := 0; i < 5000; i++ {
			chains, err := cs[0].Verify(x509.VerifyOptions{Roots: roots, Intermediates: inters, DNSName: "www.google.com", CurrentTime: time.Unix(1677615892, 0)})
			if err != nil {
				panic(err)
			}
			s += len(chains) + len(chains[0])
		}
	case "create":
		seed := make([]byte, 32)
		for i := range seed {
			seed[i] = byte(i)
		}
		k := ed25519.NewKeyFromSeed(seed)
		t := &x509.Certificate{SerialNumber: big.NewInt(1), Subject: pkix.Name{CommonName: "bench", Organization: []string{"Xo"}},
			NotBefore: time.Unix(1735689600, 0), NotAfter: time.Unix(1767225600, 0), DNSNames: []string{"bench.example"},
			KeyUsage: x509.KeyUsageDigitalSignature, ExtKeyUsage: []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth}, MaxPathLen: -1}
		for i := 0; i < 20000; i++ {
			der, err := x509.CreateCertificate(rand.Reader, t, t, k.Public(), k)
			if err != nil {
				panic(err)
			}
			s += len(der)
		}
	}
	fmt.Println(w, s)
}
