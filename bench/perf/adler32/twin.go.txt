package main

import (
	"fmt"
	"hash/adler32"
	"os"
)

func buffer(n int) []byte {
	b := make([]byte, n)
	x := 12345
	for i := range b {
		x = (x*1103515245 + 12345) % 2147483648
		b[i] = byte((x >> 16) & 255)
	}
	return b
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	big := buffer(65536)
	keys := buffer(4096)
	sum := 0
	switch w {
	case "sum":
		for i := 0; i < 4000; i++ {
			sum = sum ^ int(adler32.Checksum(big)) ^ i
		}
	case "keys":
		for i := 0; i < 2000000; i++ {
			at := i % 4000
			sum = sum ^ int(adler32.Checksum(keys[at:at+8+i%33]))
		}
	case "update":
		for i := 0; i < 2000; i++ {
			h := adler32.New()
			for at := 0; at < len(big); at += 1024 {
				h.Write(big[at : at+1024])
			}
			sum = sum ^ int(h.Sum32()) ^ i
		}
	default:
		sum = len(big) + len(keys)
	}
	fmt.Printf("%s %d\n", w, sum)
}
