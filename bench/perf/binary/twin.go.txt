package main

import (
	"encoding/binary"
	"fmt"
	"os"
)

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	sum := 0
	for i := 0; i < 1000000; i++ {
		v := i * 2654435761 % 4294967296
		switch w {
		case "fixed":
			a := binary.BigEndian.AppendUint32(nil, uint32(v))
			b := binary.LittleEndian.AppendUint64(nil, uint64(v*3))
			c := binary.BigEndian.AppendUint16(nil, uint16(v&65535))
			sum += int(binary.BigEndian.Uint32(a))%1000 + int(binary.LittleEndian.Uint64(b))%1000 + int(binary.BigEndian.Uint16(c))%1000
		case "varint":
			a := binary.AppendUvarint(nil, uint64(v))
			b := binary.AppendVarint(nil, int64(v-2147483648))
			ra, na := binary.Uvarint(a)
			rb, nb := binary.Varint(b)
			sum += int(ra)%1000 + na + int(rb)%1000 + nb
		default:
			sum += v % 3
		}
	}
	fmt.Printf("%s %d\n", w, sum)
}
