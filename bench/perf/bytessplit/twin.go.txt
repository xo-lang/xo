// The Go twin of bench/perf/bytessplit/main.xo: the same texts, separators
// and loop counts through bytes.Split, SplitN, Fields, TrimSpace,
// TrimPrefix and TrimSuffix, printing the same "<workload> <checksum>"
// line. It is a .txt so the Go tool does not build it as part of the
// repository; copy it out and `go build` it.
package main

import (
	"bytes"
	"fmt"
	"os"
)

const n = 200000

var texts = [][]byte{
	[]byte("  the quick brown fox, jumps over the lazy dog  "),
	[]byte("1,2,3,4"),
	[]byte("héllo, wörld, ça va, 日本語 テキスト   "),
	[]byte("  \t  mixed ascii and ✓ unicode ✓  "),
	[]byte("abcabcabcabcabcabcabc"),
	[]byte("a\xffb, c\x80 ,"),
}

var seps = [][]byte{[]byte(","), []byte(" "), []byte("abc"), []byte(""), []byte("✓")}
var ns = []int{2, 3, -1, 0, 5}
var affixes = [][]byte{[]byte("  "), []byte("a"), []byte("h\xc3"), []byte(""), []byte(",1")}

func splitAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range bytes.Split(texts[i%len(texts)], seps[i%len(seps)]) {
			sum += len(p)
		}
	}
	return sum
}

func splitNAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range bytes.SplitN(texts[i%len(texts)], seps[i%len(seps)], ns[i%len(ns)]) {
			sum += len(p)
		}
	}
	return sum
}

func fieldsAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range bytes.Fields(texts[i%len(texts)]) {
			sum += len(p)
		}
	}
	return sum
}

func trimAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += len(bytes.TrimSpace(texts[i%len(texts)]))
	}
	return sum
}

func affixAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		t := texts[i%len(texts)]
		p := affixes[i%len(affixes)]
		sum += len(bytes.TrimPrefix(t, p)) + len(bytes.TrimSuffix(t, p))
	}
	return sum
}

func indexAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += len(texts[i%len(texts)]) + len(seps[i%len(seps)])
	}
	return sum
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	sum := 0
	switch w {
	case "split":
		sum = splitAll()
	case "split_n":
		sum = splitNAll()
	case "fields":
		sum = fieldsAll()
	case "trim":
		sum = trimAll()
	case "affix":
		sum = affixAll()
	case "index":
		sum = indexAll()
	}
	fmt.Printf("%s %d\n", w, sum)
}
