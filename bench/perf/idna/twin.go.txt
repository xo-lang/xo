package main

import (
	"fmt"
	"os"

	"golang.org/x/net/idna"
)

func names(w string) []string {
	switch w {
	case "ascii":
		return []string{"www.example.com", "api.v2.service.internal.example.org", "a-b-c.example.net", "EXAMPLE.COM", "cdn1.static-assets.example.io", "mail.example.co.uk", "x.y.z", "localhost"}
	case "mixed":
		return []string{"bücher.example.com", "例え.jp", "правда.рф", "faß.de", "München.Straße.de", "xn--bcher-kva.example.com", "xn--80aswg.xn--p1ai", "ΕΛΛΗΝΙΚΑ.gr"}
	case "map", "enc":
		return []string{"bücher.example.com", "例え.jp", "правда.рф", "faß.de", "münchen.straße.de", "ελληνικά.gr", "café.example.org", "日本語.example.jp"}
	case "nfcmap":
		return []string{"bücher.example.com", "café.example.org", "münchen.straße.de", "århus.dk", "ñandú.es", "öl.de", "été.fr", "über.de"}
	case "dec":
		return []string{"xn--bcher-kva.example.com", "xn--r8jz45g.jp", "xn--80aafi6cg.xn--p1ai", "xn--fa-hia.de", "xn--mnchen-3ya.xn--strae-oqa.de", "xn--hxargifdar.gr", "xn--caf-dma.example.org", "xn--wgv71a119e.example.jp"}
	}
	return []string{"ééééééééééééééééééééééééééééééaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa.com", "日本語日本語日本語日本語日本語日本語日本語日本語日本語日本語日本語日本語日本語日本語日本語日本語日本語日本語日本語日本語.jp", "aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa.example.com", "ЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖЖ.рф"}
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	sum := 0
	if w != "none" {
		ns := names(w)
		rounds := 1000
		switch w {
		case "ascii":
			rounds = 20000
		case "mixed", "map", "nfcmap", "enc", "dec":
			rounds = 4000
		}
		for i := 0; i < rounds; i++ {
			for _, n := range ns {
				if w == "map" || w == "nfcmap" || w == "dec" {
					u, err := idna.Lookup.ToUnicode(n)
					if err != nil {
						u = ""
					}
					sum += len(u)
					continue
				}
				a, err := idna.Lookup.ToASCII(n)
				if err != nil {
					a = ""
				}
				sum += len(a)
				if w == "mixed" {
					u, err := idna.Lookup.ToUnicode(n)
					if err != nil {
						u = ""
					}
					sum += len(u)
				}
			}
		}
	}
	fmt.Printf("%s %d\n", w, sum)
}
