package main

import (
	"fmt"
	"os"

	"golang.org/x/crypto/blake2b"
)

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	d := make([]byte, 1<<20)
	r := 1
	for i := range d {
		r = (r*1103515245 + 12345) % 2147483648
		d[i] = byte((r >> 16) & 255)
	}
	s := 0
	switch w {
	case "big":
		for i := 0; i < 40; i++ {
			h := blake2b.Sum256(d)
			s += int(h[0]) + int(h[31])
		}
	case "big512":
		for i := 0; i < 40; i++ {
			h := blake2b.Sum512(d)
			s += int(h[0]) + int(h[31])
		}
	case "small":
		for i := 0; i < 200000; i++ {
			h := blake2b.Sum256(d[i%1000 : i%1000+i%150])
			s += int(h[0])
		}
	case "keyed":
		key := d[:32]
		for i := 0; i < 200000; i++ {
			x, err := blake2b.New(32, key)
			if err != nil {
				panic(err)
			}
			x.Write(d[i%1000 : i%1000+i%150])
			s += int(x.Sum(nil)[0])
		}
	}
	fmt.Printf("%s %d\n", w, s)
}
