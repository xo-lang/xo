// The Go twin of bench/perf/strsplit/main.xo: the same texts, separators
// and loop counts through strings.Split and strings.Fields, printing the
// same "<workload> <checksum> <elapsed us>" line. It is a .txt so the Go
// tool does not build it as part of the repository; copy it out and
// `go build` it.
package main

import (
	"bytes"
	"fmt"
	"os"
	"strings"
	"time"
)

const n = 200000

var texts = []string{
	"the,quick,brown,fox,jumps,over,the,lazy,dog",
	"1,2,3,4",
	"héllo, wörld, ça va, 日本語 テキスト   ",
	"  \t  mixed ascii and ✓ unicode ✓  ",
	"abcabcabcabcabcabcabc",
	"the quick brown fox jumps over the lazy dog",
	"a�b, c� ,",
	"",
	",,,",
}

var seps = []string{",", " ", "abc", "", "✓"}

var linesTexts = []string{
	"alpha\nbeta\ngamma\n",
	"one\r\ntwo\r\nthree",
	"\n\n\n",
	"no newline here",
	"héllo\nwörld\n",
	"a\n",
}

func splitAfterAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range strings.SplitAfter(texts[i%len(texts)], ",") {
			sum += len(p)
		}
	}
	return sum
}

// lines is Str.lines: split on \n, a trailing \r of each piece removed,
// and no empty last piece after a final newline. Sized the way the Xo
// version sizes its list, with the newline count plus one.
func lines(s string) []string {
	out := make([]string, 0, strings.Count(s, "\n")+1)
	start := 0
	for i := 0; i < len(s); i++ {
		if s[i] == '\n' {
			end := i
			if end > start && s[end-1] == '\r' {
				end--
			}
			out = append(out, s[start:end])
			start = i + 1
		}
	}
	if start < len(s) {
		end := len(s)
		if s[end-1] == '\r' {
			end--
		}
		out = append(out, s[start:end])
	}
	return out
}

func linesAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range lines(linesTexts[i%len(linesTexts)]) {
			sum += len(p)
		}
	}
	return sum
}

func bytesSplitAll() int {
	sep := []byte(",")
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range bytes.Split([]byte(texts[i%len(texts)]), sep) {
			sum += len(p)
		}
	}
	return sum
}

func bytesFieldsAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range bytes.Fields([]byte(texts[i%len(texts)])) {
			sum += len(p)
		}
	}
	return sum
}

func splitAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range strings.Split(texts[i%len(texts)], seps[i%len(seps)]) {
			sum += len(p)
		}
	}
	return sum
}

func splitOne() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range strings.Split(texts[i%len(texts)], ",") {
			sum += len(p)
		}
	}
	return sum
}

func splitMulti() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range strings.Split(texts[i%len(texts)], "abc") {
			sum += len(p)
		}
	}
	return sum
}

func splitAbsent() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range strings.Split(texts[i%len(texts)], "§") {
			sum += len(p)
		}
	}
	return sum
}

func splitEmpty() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range strings.Split(texts[i%len(texts)], "") {
			sum += len(p)
		}
	}
	return sum
}

func fieldsAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		for _, p := range strings.Fields(texts[i%len(texts)]) {
			sum += len(p)
		}
	}
	return sum
}

func indexAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += len(texts[i%len(texts)]) + len(seps[i%len(seps)])
	}
	return sum
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	t := time.Now()
	sum := 0
	switch w {
	case "split":
		sum = splitAll()
	case "split_one":
		sum = splitOne()
	case "split_multi":
		sum = splitMulti()
	case "split_absent":
		sum = splitAbsent()
	case "split_empty":
		sum = splitEmpty()
	case "split_after":
		sum = splitAfterAll()
	case "lines":
		sum = linesAll()
	case "bytes_split":
		sum = bytesSplitAll()
	case "bytes_fields":
		sum = bytesFieldsAll()
	case "fields":
		sum = fieldsAll()
	case "index":
		sum = indexAll()
	}
	us := time.Since(t).Microseconds()
	fmt.Println(w, sum, us)
}
