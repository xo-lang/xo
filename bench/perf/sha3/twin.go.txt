package main

import (
	"crypto/sha3"
	"fmt"
	"os"
)

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	d := make([]byte, 1<<20)
	r := 1
	for i := range d {
		r = (r*1103515245 + 12345) % 2147483648
		d[i] = byte((r >> 16) & 255)
	}
	s := 0
	if w == "big" {
		for i := 0; i < 40; i++ {
			h := sha3.Sum256(d)
			s += int(h[0]) + int(h[31])
		}
	} else if w == "small" {
		for i := 0; i < 200000; i++ {
			h := sha3.Sum256(d[i%1000 : i%1000+i%150])
			s += int(h[0])
		}
	}
	fmt.Printf("%s %d\n", w, s)
}
