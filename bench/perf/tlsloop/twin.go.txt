// Go twin of tlsloop/main.xo: a net/http server over TLS 1.3 and its
// client in one process, N POSTs to /echo on one HTTP/2 connection:
// std/http offers ALPN h2 first on both sides over TLS, as net/http's
// defaults do (ForceAttemptHTTP2 because the Transport has its own
// TLSClientConfig); the first response must be HTTP/2. Until
// 2026-11-19 this twin turned h2 off on both sides, on the wrong belief
// that Xo's ALPN left it out, and timed HTTP/1.1 against Xo's HTTP/2.
// Reads cert.pem, key.pem and ca.pem from the
// working directory. Prints `<workload> <bytes received>`.
package main

import (
	"bytes"
	"crypto/tls"
	"crypto/x509"
	"fmt"
	"io"
	"net"
	"net/http"
	"os"
	"strings"
)

const n = 20000

func main() {
	w := "small"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	size := 16
	if w == "big" {
		size = 16384
	}
	body := []byte(strings.Repeat("x", size))
	cert, err := tls.LoadX509KeyPair("cert.pem", "key.pem")
	if err != nil {
		panic(err)
	}
	ca, err := os.ReadFile("ca.pem")
	if err != nil {
		panic(err)
	}
	roots := x509.NewCertPool()
	if !roots.AppendCertsFromPEM(ca) {
		panic("roots")
	}
	mux := http.NewServeMux()
	mux.HandleFunc("/echo", func(rw http.ResponseWriter, r *http.Request) {
		b, _ := io.ReadAll(r.Body)
		rw.Header().Set("Content-Type", "text/plain; charset=utf-8")
		rw.WriteHeader(200)
		rw.Write(b)
	})
	l, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		panic(err)
	}
	srv := &http.Server{
		Handler:   mux,
		TLSConfig: &tls.Config{Certificates: []tls.Certificate{cert}, MinVersion: tls.VersionTLS13},
	}
	go srv.ServeTLS(l, "", "")
	c := &http.Client{Transport: &http.Transport{
		TLSClientConfig:   &tls.Config{RootCAs: roots, MinVersion: tls.VersionTLS13},
		ForceAttemptHTTP2: true,
	}}
	url := "https://" + l.Addr().String() + "/echo"
	total := 0
	for i := 0; i < n; i++ {
		resp, err := c.Post(url, "text/plain; charset=utf-8", bytes.NewReader(body))
		if err != nil {
			panic(err)
		}
		if resp.ProtoMajor != 2 {
			panic("tlsloop twin: " + resp.Proto + ", not HTTP/2")
		}
		b, err := io.ReadAll(resp.Body)
		resp.Body.Close()
		if err != nil {
			panic(err)
		}
		total += len(b)
	}
	srv.Close()
	fmt.Printf("%s %d\n", w, total)
}
