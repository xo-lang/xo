// The Go twin of bench/perf/trimchars/main.xo: the same texts, cutsets and
// loop counts through strings.Trim, TrimLeft and TrimRight, printing the
// same "<workload> <checksum>" line. It is a .txt so the Go tool does not
// build it as part of the repository; copy it out and `go build` it.
package main

import (
	"fmt"
	"os"
	"strings"
)

const n = 200000

var texts = []string{
	"  the quick brown fox jumps over the lazy dog  ",
	"***********  a  ***********",
	"héllo, wörld, ça va, 日本語 テキスト   ",
	"  \t  mixed ascii and ✓ unicode ✓  ",
	"abcabcabcabcabcabcabc",
	"   ",
}

var cutsets = []string{" ", "*", " \t", "é✓", "abc", "", "!"}

func trimAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += len(strings.Trim(texts[i%len(texts)], cutsets[i%len(cutsets)]))
	}
	return sum
}

func trimStartAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += len(strings.TrimLeft(texts[i%len(texts)], cutsets[i%len(cutsets)]))
	}
	return sum
}

func trimEndAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += len(strings.TrimRight(texts[i%len(texts)], cutsets[i%len(cutsets)]))
	}
	return sum
}

// indexAll is the loop's own cost, the two table lookups and the
// lengths, with no trim.
func indexAll() int {
	sum := 0
	for i := 0; i < n; i++ {
		sum += len(texts[i%len(texts)]) + len(cutsets[i%len(cutsets)])
	}
	return sum
}

func main() {
	w := "none"
	if len(os.Args) > 1 {
		w = os.Args[1]
	}
	sum := 0
	switch w {
	case "trim":
		sum = trimAll()
	case "trim_start":
		sum = trimStartAll()
	case "trim_end":
		sum = trimEndAll()
	case "index":
		sum = indexAll()
	}
	fmt.Printf("%s %d\n", w, sum)
}
