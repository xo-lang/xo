// dump prints the table of std/idna (scripts/genidna copies this file next to
// x/net's idna tables, trie and trieval, so it can read their internals).
package idna

import (
	"fmt"
	"os"
	"strings"

	"golang.org/x/text/unicode/bidi"
)

func varint(b []byte, v int) []byte {
	for v >= 0x80 {
		b = append(b, byte(v&0x7f|0x80))
		v >>= 7
	}
	return append(b, byte(v))
}

func zigzag(d int) int {
	if d < 0 {
		return -2*d - 1
	}
	return 2 * d
}

type rec struct {
	cat, flags, class int
	kind              int // 0 none, 1 delta, 2 string
	delta             int
	str               []rune
}

func (a rec) same(b rec) bool {
	if a.cat != b.cat || a.flags != b.flags || a.class != b.class || a.kind != b.kind || a.delta != b.delta || len(a.str) != len(b.str) {
		return false
	}
	for i := range a.str {
		if a.str[i] != b.str[i] {
			return false
		}
	}
	return true
}

func mappingOf(v info, r rune) []byte {
	return v.appendMapping(nil, string(r))
}

// from x/net/idna's idna.go (BSD-style license, Copyright 2016 The Go Authors)
func (c info) isBidi(s string) bool {
	if !c.isMapped() {
		return c&attributesMask == rtl
	}
	p, _ := bidi.LookupString(s)
	switch p.Class() {
	case bidi.R, bidi.AL, bidi.AN:
		return true
	}
	return false
}

func (c info) appendMapping(b []byte, s string) []byte {
	index := int(c >> indexShift)
	if c&xorBit == 0 {
		p := index
		return append(b, mappings[mappingIndex[p]:mappingIndex[p+1]]...)
	}
	b = append(b, s...)
	if c&inlineXOR == inlineXOR {
		b[len(b)-1] ^= byte(index)
	} else {
		for p := len(b) - int(xorData[index]); p < len(b); p++ {
			index++
			b[p] ^= xorData[index]
		}
	}
	return b
}

func Dump(path string) {
	var recs, maps []byte
	var cur rec
	start := 0
	var haveCur bool
	nrec := 0
	flush := func() {
		recs = append(recs, byte(start), byte(start>>8), byte(start>>16), byte(cur.cat), byte(cur.flags), byte(cur.class))
		pos := 0
		if cur.kind != 0 {
			pos = len(maps) + 1
			switch cur.kind {
			case 1:
				maps = varint(maps, 0)
				maps = varint(maps, zigzag(cur.delta))
			case 2:
				maps = varint(maps, len(cur.str)+1)
				for _, m := range cur.str {
					maps = varint(maps, int(m))
				}
			}
		}
		recs = append(recs, byte(pos), byte(pos>>8), byte(pos>>16))
		nrec++
	}
	for r := rune(0); r <= 0x10FFFF; r++ {
		var x rec
		if r >= 0xD800 && r <= 0xDFFF {
			x = rec{cat: int(disallowed)}
		} else {
			v, sz := trie.lookupString(string(r))
			if sz == 0 {
				panic(fmt.Sprintf("no trie value for %U", r))
			}
			info := info(v)
			x.cat = int(info.category())
			join := 0
			if !info.isMapped() {
				join = int(info.joinType())
			}
			attr := 0
			if !info.isMapped() {
				switch info & attributesMask {
				case rtl:
					attr = 1
				case modifier:
					attr = 2
				case viramaModifier:
					attr = 3
				}
			}
			may := 0
			if info&mayNeedNorm != 0 {
				may = 1
			}
			isb := 0
			if info.isBidi(string(r)) {
				isb = 1
			}
			x.flags = join | attr<<3 | may<<5 | isb<<6
			p, _ := bidi.LookupRune(r)
			x.class = int(p.Class())
			if info.isMapped() {
				mb := mappingOf(info, r)
				ms := []rune(string(mb))
				if len(ms) == 1 && string(mb) != "\ufffd" {
					x.kind = 1
					x.delta = int(ms[0]) - int(r)
				} else {
					x.kind = 2
					x.str = ms
				}
			}
		}
		if !haveCur {
			cur, haveCur, start = x, true, int(r)
		} else if !cur.same(x) {
			flush()
			cur, start = x, int(r)
		}
	}
	flush()
	var w strings.Builder
	fmt.Fprintf(&w, "// Unicode %s, from golang.org/x/net/idna's tables (scripts/genidna): %d ranges of runes (%d bytes) and %d bytes of mappings.\n", UnicodeVersion, nrec, len(recs), len(maps))
	fmt.Fprintf(&w, "const UNICODE_VERSION: Str = %q\n\n", UnicodeVersion)
	w.WriteString("/// Per range of runes, 9 bytes, ascending: its first rune (3 bytes), the category, the flags (join type, attributes, may need norm, bidi), the bidi class, and the position plus one of its mapping in MAPS (3 bytes, 0 for none).\nconst RANGES: Bytes = " + bytesLit(recs) + "\n\n")
	w.WriteString("/// Mappings: 0 and a zigzag varint delta from the rune, or a count plus 1 and that many varint runes.\nconst MAPS: Bytes = " + bytesLit(maps) + "\n\n")
	b, err := os.ReadFile(path)
	if err != nil {
		panic(err)
	}
	src := string(b)
	i := strings.Index(src, "// BEGIN GENERATED\n")
	j := strings.Index(src, "// END GENERATED\n")
	if i < 0 || j < 0 {
		panic("markers missing")
	}
	if err := os.WriteFile(path, []byte(src[:i]+"// BEGIN GENERATED\n"+w.String()+src[j:]), 0o644); err != nil {
		panic(err)
	}
}

func bytesLit(b []byte) string {
	var w strings.Builder
	w.WriteString(`b"`)
	for _, c := range b {
		fmt.Fprintf(&w, `\x%02x`, c)
	}
	w.WriteString(`"`)
	return w.String()
}
